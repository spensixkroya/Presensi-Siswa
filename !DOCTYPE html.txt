<script>
  const SCRIPT_URL = "URL_WEB_APP_ANDA"; // Ganti dengan URL Google Apps Script
  let sedangProses = false;

  // Efek suara beep saat berhasil scan
  function bunyiBeep() {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = ctx.createOscillator();
    osc.type = "sine";
    osc.frequency.setValueAtTime(800, ctx.currentTime);
    osc.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + 0.15);
  }

  function onScanSuccess(decodedText) {
    if (sedangProses) return; // Cegah scan ganda beruntun
    sedangProses = true;
    bunyiBeep();

    const boxHasil = document.getElementById("hasil");
    boxHasil.style.display = "block";
    boxHasil.className = "";
    boxHasil.style.background = "#fff3cd";
    boxHasil.style.color = "#856404";
    boxHasil.innerText = "Memproses NISN: " + decodedText + "...";

    const formData = new FormData();
    formData.append("nisn", decodedText);

    fetch(SCRIPT_URL, { method: "POST", body: formData })
      .then(res => res.json())
      .then(data => {
        if (data.status === "success") {
          boxHasil.className = "sukses";
          boxHasil.innerHTML = `✅ ${data.nama} (${data.kelas})<br>Jam: ${data.waktu} — ${data.keterangan}`;
        } else {
          boxHasil.className = "gagal";
          boxHasil.innerText = "❌ " + data.message;
        }
      })
      .catch(() => {
        boxHasil.className = "gagal";
        boxHasil.innerText = "⚠️ Gagal terhubung ke server!";
      })
      .finally(() => {
        // Jeda 3 detik sebelum siap memindai kartu berikutnya
        setTimeout(() => { sedangProses = false; }, 3000);
      });
  }

  const html5QrcodeScanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: 250 });
  html5QrcodeScanner.render(onScanSuccess);
</script>
